\chapter{Introduction}
% ============================================================

This report presents a comprehensive statistical analysis of \stat{129,780} airline flight records. The objective is to reveal the key factors influencing coach ticket prices. A reproducible sample of \stat{10,000} observations is drawn using Student ID \texttt{12310048} as the random seed.

\begin{lstlisting}[style=Rcode, caption={R — Load dataset and draw 10{,}000-row reproducible sample}]
df_full <- read.csv("flight_data.csv", stringsAsFactors = FALSE)
names(df_full)[1] <- "miles"

set.seed(12310048)
idx <- sample(nrow(df_full), 10000)

\end{lstlisting}

from scipy.stats import norm
import pandas as pd

cp = df["coach_price"]

# Z-test (H0: mu = $350)
z = (cp.mean() - 350) / (cp.std() / np.sqrt(len(cp)))
p_z = 2 * (1 - norm.cdf(abs(z)))
print(f"Z-test: z={z:.4f}  p={p_z:.4e}")

# One-sample t-test (H0: mu = $380)
t, p_t = stats.ttest_1samp(cp, 380)
print(f"t-test: t={t:.4f}  p={p_t:.4e}")

# Chi-square tests
for a, b in [("redeye","inflight_meal"), ("redeye","weekend")]:
    ct = pd.crosstab(df[a], df[b])
    chi2, p_chi, dof, _ = stats.chi2_contingency(ct)
    print(f"Chi-sq {a}x{b}: chi2={chi2:.4f}  p={p_chi:.4f}")
